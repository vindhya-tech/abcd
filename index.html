Experiment-2- GAN

import tensorflow as tf
from tensorflow.keras import layers
import numpy as np
import matplotlib.pyplot as plt
import os
import time

# ------------------------------------
# 1. Configuration Parameters
# ------------------------------------
LATENT_DIM = 100
BATCH_SIZE = 128
EPOCHS = 50
BUFFER_SIZE = 60000
OUTPUT_DIR = "gan_images"

os.makedirs(OUTPUT_DIR, exist_ok=True)

# ------------------------------------
# 2. Set Random Seeds
# ------------------------------------
tf.random.set_seed(42)
np.random.seed(42)

# ------------------------------------
# 3. Load MNIST Dataset
# ------------------------------------
(x_train, _), (_, _) = tf.keras.datasets.mnist.load_data()

# ------------------------------------
# 4. Preprocess Data
# ------------------------------------
x_train = x_train.astype("float32")
x_train = (x_train - 127.5) / 127.5 # Normalize to [-1,1]
x_train = np.expand_dims(x_train, axis=-1) # (28,28,1)

# ------------------------------------
# 5. Create Dataset Pipeline
# ------------------------------------
dataset = tf.data.Dataset.from_tensor_slices(x_train)
dataset = dataset.shuffle(BUFFER_SIZE).batch(BATCH_SIZE)

# ------------------------------------
# 6. Build Generator
# ------------------------------------
def build_generator():
model = tf.keras.Sequential()

model.add(layers.Dense(7*7*256, use_bias=False, input_shape=(LATENT_DIM,)))
model.add(layers.BatchNormalization())
model.add(layers.LeakyReLU())

model.add(layers.Reshape((7, 7, 256)))

model.add(layers.Conv2DTranspose(128, (5,5), strides=(1,1), padding='same'))
model.add(layers.BatchNormalization())
model.add(layers.LeakyReLU())

model.add(layers.Conv2DTranspose(64, (5,5), strides=(2,2), padding='same'))
model.add(layers.BatchNormalization())
model.add(layers.LeakyReLU())

model.add(layers.Conv2DTranspose(1, (5,5), strides=(2,2),
padding='same', activation='tanh'))

return model

generator = build_generator()

# ------------------------------------
# 7. Build Discriminator
# ------------------------------------
def build_discriminator():
model = tf.keras.Sequential()

model.add(layers.Conv2D(64, (5,5), strides=(2,2),
padding='same', input_shape=[28,28,1]))
model.add(layers.LeakyReLU())
model.add(layers.Dropout(0.3))

model.add(layers.Conv2D(128, (5,5), strides=(2,2), padding='same'))
model.add(layers.LeakyReLU())
model.add(layers.Dropout(0.3))

model.add(layers.Flatten())
model.add(layers.Dense(1))

return model

discriminator = build_discriminator()

# ------------------------------------
# 8. Loss Function
# ------------------------------------
loss_fn = tf.keras.losses.BinaryCrossentropy(from_logits=True)

def generator_loss(fake_output):
return loss_fn(tf.ones_like(fake_output), fake_output)

def discriminator_loss(real_output, fake_output):
real_loss = loss_fn(tf.ones_like(real_output), real_output)
fake_loss = loss_fn(tf.zeros_like(fake_output), fake_output)
return real_loss + fake_loss

# ------------------------------------
# 9. Optimizers
# ------------------------------------
gen_optimizer = tf.keras.optimizers.Adam(0.0002, beta_1=0.5)
disc_optimizer = tf.keras.optimizers.Adam(0.0002, beta_1=0.5)

# ------------------------------------
# 10. Fixed Noise Vector
# ------------------------------------
NUM_EXAMPLES = 16
seed = tf.random.normal([NUM_EXAMPLES, LATENT_DIM])

# ------------------------------------
# 11. Training Step
# ------------------------------------
@tf.function
def train_step(images):

noise = tf.random.normal([BATCH_SIZE, LATENT_DIM])

with tf.GradientTape() as gen_tape, tf.GradientTape() as disc_tape:

fake_images = generator(noise, training=True)

real_output = discriminator(images, training=True)
fake_output = discriminator(fake_images, training=True)

gen_loss = generator_loss(fake_output)
disc_loss = discriminator_loss(real_output, fake_output)

gen_gradients = gen_tape.gradient(gen_loss, generator.trainable_variables)
disc_gradients = disc_tape.gradient(disc_loss, discriminator.trainable_variables)

gen_optimizer.apply_gradients(zip(gen_gradients,
generator.trainable_variables))
disc_optimizer.apply_gradients(zip(disc_gradients,
discriminator.trainable_variables))

return gen_loss, disc_loss

# ------------------------------------
# 12. Save Generated Images
# ------------------------------------
def generate_and_save_images(model, epoch, test_input):

predictions = model(test_input, training=False)

fig = plt.figure(figsize=(4,4))

for i in range(predictions.shape[0]):
plt.subplot(4,4,i+1)
plt.imshow((predictions[i,:,:,0] + 1)/2, cmap='gray')
plt.axis('off')

plt.savefig(f"{OUTPUT_DIR}/epoch_{epoch}.png")
plt.close()

# ------------------------------------
# 13. Training Loop
# ------------------------------------
def train(dataset, epochs):

for epoch in range(epochs):
start = time.time()

for image_batch in dataset:
gen_loss, disc_loss = train_step(image_batch)

generate_and_save_images(generator, epoch+1, seed)

print(f"Epoch {epoch+1}/{epochs} | "
f"Gen Loss: {gen_loss:.4f} | "
f"Disc Loss: {disc_loss:.4f} | "
f"Time: {time.time()-start:.2f}s")

# ------------------------------------
# 14. Start Training
# ------------------------------------
train(dataset, EPOCHS)

# ------------------------------------
# 15. Generate New Images After Training
# ------------------------------------
random_noise = tf.random.normal([16, LATENT_DIM])
generate_and_save_images(generator, "final", random_noise)

print("Training completed.")





Experiment-3- VAE

import numpy as np
import matplotlib.pyplot as plt
import tensorflow as tf
from tensorflow import keras
from tensorflow.keras import layers, ops

# -----------------------------
# Data Loading & Preprocessing
# -----------------------------
(x_train, _), (x_test, _) = keras.datasets.fashion_mnist.load_data()

x_train = x_train.astype("float32") / 255.0
x_test = x_test.astype("float32") / 255.0

x_train = x_train.reshape(-1, 784)
x_test = x_test.reshape(-1, 784)

LATENT_DIM = 16
EPS = 1e-7

# -----------------------------
# Reparameterization Trick
# -----------------------------
def sample_z(args):
z_mean, z_log_var = args
eps = keras.random.normal(shape=ops.shape(z_mean))
return z_mean + ops.exp(0.5 * z_log_var) * eps

# -----------------------------
# Encoder
# -----------------------------
inp = keras.Input(shape=(784,))
h = layers.Dense(256, activation="relu")(inp)
h = layers.Dense(128, activation="relu")(h)

z_mean = layers.Dense(LATENT_DIM, name="z_mean")(h)
z_log_var = layers.Dense(LATENT_DIM, name="z_log_var")(h)
z = layers.Lambda(sample_z, name="z")([z_mean, z_log_var])

# -----------------------------
# Decoder
# -----------------------------
d = layers.Dense(128, activation="relu")(z)
d = layers.Dense(256, activation="relu")(d)
recon = layers.Dense(784, activation="sigmoid", name="recon")(d)

# Pack outputs (as shown in slides)
out = layers.Concatenate()([recon, z_mean, z_log_var])

vae = keras.Model(inp, out)

# -----------------------------
# Loss Functions
# -----------------------------
def recon_bce_per_sample(y_true, y_pred):
y_pred = ops.clip(y_pred, EPS, 1.0 - EPS)
bce = -(y_true * ops.log(y_pred) +
(1.0 - y_true) * ops.log(1.0 - y_pred))
return ops.sum(bce, axis=1)

def kl_per_sample(z_mean, z_log_var):
return -0.5 * ops.sum(
1 + z_log_var - ops.square(z_mean) - ops.exp(z_log_var),
axis=1
)

def vae_loss(y_true, y_pred):
recon = y_pred[:, :784]
z_mean_p = y_pred[:, 784:784 + LATENT_DIM]
z_log_var_p = y_pred[:, 784 + LATENT_DIM:]

recon_loss = recon_bce_per_sample(y_true, recon)
kl_loss = kl_per_sample(z_mean_p, z_log_var_p)

return ops.mean(recon_loss + kl_loss)

# -----------------------------
# Compile & Train
# -----------------------------
vae.compile(optimizer="adam", loss=vae_loss)

vae.fit(
x_train, x_train,
validation_data=(x_test, x_test),
epochs=20,
batch_size=128
)

# -----------------------------
# Evaluation
# -----------------------------
vae.evaluate(x_test, x_test)

# -----------------------------
# Reconstruction & Plotting
# -----------------------------
idx = np.random.randint(0, x_test.shape[0])
x_in = x_test[idx:idx + 1]

packed_out = vae.predict(x_in, verbose=0)
recon_flat = packed_out[:, :784]

orig_img = x_in.reshape(28, 28)
recon_img = recon_flat.reshape(28, 28)

plt.figure(figsize=(6, 3))

plt.subplot(1, 2, 1)
plt.imshow(orig_img, cmap="gray")
plt.title("Original Image")
plt.axis("off")

plt.subplot(1, 2, 2)
plt.imshow(recon_img, cmap="gray")
plt.title("Reconstructed Image")
plt.axis("off")

plt.tight_layout()
plt.show()



Experiment-4-Image captioning

import os
import re
import numpy as np
import pandas as pd
import tensorflow as tf
from tensorflow.keras import layers, Model
from tensorflow.keras.applications import ResNet50
from tensorflow.keras.applications.resnet50 import preprocess_input
from tensorflow.keras.preprocessing.image import load_img, img_to_array
from tensorflow.keras.preprocessing.text import Tokenizer
from tensorflow.keras.preprocessing.sequence import pad_sequences

IMAGE_DIR = "Images"
CAPTION_FILE = "captions.txt"

MAX_IMAGES = 300
MAX_LEN = 25
VOCAB_SIZE = 3000
EMBED_DIM = 128
BATCH_SIZE = 32
EPOCHS = 5

data = pd.read_csv(CAPTION_FILE)

print(data.head())
print(data.columns)

def clean_text(text):
text = str(text).lower()
text = re.sub(r"[^a-z ]", "", text)
text = re.sub(r"\s+", " ", text).strip()
return "startseq " + text + " endseq"

captions = {}

for _, row in data.iterrows():
img = row["image"]
cap = clean_text(row["caption"])

if len(captions) >= MAX_IMAGES and img not in captions:
continue

img_path = os.path.join(IMAGE_DIR, img)

if os.path.exists(img_path):
captions.setdefault(img, []).append(cap)

print("Total images used:", len(captions))

all_captions = []

for caps in captions.values():
all_captions.extend(caps)

tokenizer = Tokenizer(num_words=VOCAB_SIZE, oov_token="")
tokenizer.fit_on_texts(all_captions)

vocab_size = min(VOCAB_SIZE, len(tokenizer.word_index)) + 1

base_model = ResNet50(weights="imagenet", include_top=False)
base_model.trainable = False

cnn_model = Model(base_model.input, base_model.output)

def extract_feature(img_name):
path = os.path.join(IMAGE_DIR, img_name)

img = load_img(path, target_size=(224, 224))
img = img_to_array(img)
img = np.expand_dims(img, axis=0)
img = img.copy()
img = preprocess_input(img)

feature = cnn_model.predict(img, verbose=0)
feature = feature.reshape(49, 2048)

return feature

features = {}

for img_name in captions.keys():
features[img_name] = extract_feature(img_name)

X_img = []
X_seq = []
Y = []

for img_name, caps in captions.items():
feature = features[img_name]

for cap in caps:
seq = tokenizer.texts_to_sequences([cap])[0]

for i in range(1, len(seq)):
input_seq = pad_sequences(
[seq[:i]],
maxlen=MAX_LEN,
padding="post"
)[0]

X_img.append(feature)
X_seq.append(input_seq)
Y.append(seq[i])

X_img = np.array(X_img, dtype=np.float32)
X_seq = np.array(X_seq, dtype=np.int32)
Y = np.array(Y, dtype=np.int32)

print("Training samples:", len(Y))

image_input = layers.Input(shape=(49, 2048))
caption_input = layers.Input(shape=(MAX_LEN,))

image_features = layers.Dense(EMBED_DIM)(image_input)

caption_embedding = layers.Embedding(
vocab_size,
EMBED_DIM
)(caption_input)

attention = layers.MultiHeadAttention(
num_heads=2,
key_dim=64
)(caption_embedding, image_features)

x = layers.Add()([caption_embedding, attention])
x = layers.LayerNormalization()(x)
x = layers.GlobalAveragePooling1D()(x)

output = layers.Dense(vocab_size, activation="softmax")(x)

model = Model([image_input, caption_input], output)

model.compile(
optimizer="adam",
loss="sparse_categorical_crossentropy",
metrics=["accuracy"]
)

model.summary()

model.fit(
[X_img, X_seq],
Y,
batch_size=BATCH_SIZE,
epochs=EPOCHS
)

def generate_caption(img_name):
feature = features[img_name]
caption = ["startseq"]

for _ in range(MAX_LEN):
seq = tokenizer.texts_to_sequences([" ".join(caption)])[0]
seq = pad_sequences([seq], maxlen=MAX_LEN, padding="post")

pred = model.predict(
[feature.reshape(1, 49, 2048), seq],
verbose=0
)

pred[0][0] = -1

word_id = np.argmax(pred)
word = tokenizer.index_word.get(word_id, "")

if word == "endseq" or word == "":
break

caption.append(word)

return " ".join(caption[1:])

test_img = list(features.keys())[0]

print("Test Image:", test_img)
print("Generated Caption:")
print(generate_caption(test_img))





Experiment-5- Seq-to-seq

import numpy as np
import pandas as pd
import re
import tensorflow as tf
from tensorflow.keras.preprocessing.text import Tokenizer
from tensorflow.keras.preprocessing.sequence import pad_sequences
from tensorflow.keras.layers import Input, LSTM, Embedding, Dense
from tensorflow.keras.layers import AdditiveAttention, Concatenate
from tensorflow.keras.models import Model

# LOAD DATA
data = pd.read_csv("custom_summarization_dataset.csv")

texts = data["text"]
summaries = data["summary"]

# CLEAN TEXT
def clean_text(text):
text = text.lower()
text = re.sub(r"[^a-zA-Z ]", "", text)
return text

texts = texts.apply(clean_text)
summaries = summaries.apply(clean_text)

# ADD START AND END TOKENS
summaries = summaries.apply(lambda x: "startseq " + x + " endseq")

# TOKENIZATION
tokenizer = Tokenizer()
tokenizer.fit_on_texts(texts.tolist() + summaries.tolist())

vocab_size = len(tokenizer.word_index) + 1

text_seq = tokenizer.texts_to_sequences(texts)
summary_seq = tokenizer.texts_to_sequences(summaries)

max_text_len = 100
max_summary_len = 20

encoder_input = pad_sequences(text_seq, maxlen=max_text_len, padding="post")
decoder_input = pad_sequences(summary_seq, maxlen=max_summary_len, padding="post")

decoder_target = np.zeros_like(decoder_input)
decoder_target[:, :-1] = decoder_input[:, 1:]
decoder_target = np.expand_dims(decoder_target, -1)

# ENCODER
encoder_inputs = Input(shape=(max_text_len,))
enc_emb = Embedding(vocab_size, 128)(encoder_inputs)

encoder_outputs, state_h, state_c = LSTM(
128,
return_sequences=True,
return_state=True
)(enc_emb)

# DECODER
decoder_inputs = Input(shape=(max_summary_len,))
dec_emb = Embedding(vocab_size, 128)(decoder_inputs)

decoder_outputs, _, _ = LSTM(
128,
return_sequences=True,
return_state=True
)(dec_emb, initial_state=[state_h, state_c])

# ATTENTION
attention = AdditiveAttention()
attention_output = attention([decoder_outputs, encoder_outputs])

concat = Concatenate(axis=-1)([decoder_outputs, attention_output])

# OUTPUT
outputs = Dense(vocab_size, activation="softmax")(concat)

# MODEL
model = Model([encoder_inputs, decoder_inputs], outputs)

model.compile(
optimizer="adam",
loss="sparse_categorical_crossentropy",
metrics=["accuracy"]
)

# TRAIN
model.fit(
[encoder_input, decoder_input],
decoder_target,
epochs=50,
batch_size=16,
validation_split=0.2
)

# TEST SUMMARY
test_text = """
Artificial intelligence is a branch of computer science
that focuses on building smart machines.
"""

test_text = clean_text(test_text)

seq = tokenizer.texts_to_sequences([test_text])
seq = pad_sequences(seq, maxlen=max_text_len, padding="post")

summary_seq = [tokenizer.word_index["startseq"]]

reverse_word_index = tokenizer.index_word

generated_summary = []
used_words = set()

for i in range(max_summary_len):

dec_input = pad_sequences(
[summary_seq],
maxlen=max_summary_len,
padding="post"
)

pred = model.predict([seq, dec_input], verbose=0)

preds_step = pred[0, i, :]

preds_step[0] = -1 # ignore padding

# avoid repeated words
for word in used_words:
word_id = tokenizer.word_index.get(word)
if word_id is not None:
preds_step[word_id] = -1

next_word_id = np.argmax(preds_step)
word = reverse_word_index.get(next_word_id, "")

if word == "endseq" or word == "":
break

if word != "startseq":
generated_summary.append(word)
used_words.add(word)

summary_seq.append(next_word_id)

print("Generated Summary:")
print(" ".join(generated_summary))

Experiment-7-Captioning

import os
import re
import numpy as np
import pandas as pd
import matplotlib.pyplot as plt
from PIL import Image

import tensorflow as tf
from tensorflow.keras import layers, Model
from tensorflow.keras.applications import ResNet50
from tensorflow.keras.applications.resnet50 import preprocess_input
from tensorflow.keras.preprocessing.image import load_img, img_to_array
from tensorflow.keras.preprocessing.text import Tokenizer
from tensorflow.keras.preprocessing.sequence import pad_sequences

IMAGE_DIR = "Images"
CAPTION_FILE = "captions.txt"

MAX_IMAGES = 500
MAX_LEN = 25
VOCAB_SIZE = 3000
EMBED_DIM = 128
LSTM_UNITS = 256
BATCH_SIZE = 64
EPOCHS = 5
IMG_SIZE = (224, 224)

df = pd.read_csv(CAPTION_FILE)

def clean_caption(text):
text = text.lower()
text = re.sub(r"[^a-z\s]", "", text)
text = re.sub(r"\s+", " ", text).strip()
return "startseq " + text + " endseq"

captions = {}

for _, row in df.iterrows():
img = row["image"]
cap = clean_caption(row["caption"])

if len(captions) >= MAX_IMAGES and img not in captions:
continue

captions.setdefault(img, []).append(cap)

all_captions = []
for caps in captions.values():
all_captions.extend(caps)

tokenizer = Tokenizer(num_words=VOCAB_SIZE, oov_token="")
tokenizer.fit_on_texts(all_captions)

vocab_size = min(VOCAB_SIZE, len(tokenizer.word_index) + 1)

print("Total images:", len(captions))
print("Vocabulary size:", vocab_size)

base_model = ResNet50(weights="imagenet", include_top=False, input_shape=(224, 224, 3))
base_model.trainable = False

x = layers.GlobalAveragePooling2D()(base_model.output)
feature_extractor = Model(base_model.input, x)

def extract_features(image_name):
path = os.path.join(IMAGE_DIR, image_name)
img = load_img(path, target_size=IMG_SIZE)
img = img_to_array(img)
img = np.expand_dims(img, axis=0)
img = preprocess_input(img)
feature = feature_extractor.predict(img, verbose=0)
return feature[0]

features = {}

for img_name in captions.keys():
try:
features[img_name] = extract_features(img_name)
except:
print("Skipped:", img_name)

print("Extracted features:", len(features))

X_img = []
X_seq = []
y = []

for img_name, caps in captions.items():
if img_name not in features:
continue

feature = features[img_name]

for cap in caps:
seq = tokenizer.texts_to_sequences([cap])[0]

for i in range(1, len(seq)):
in_seq = seq[:i]
out_word = seq[i]

in_seq = pad_sequences([in_seq], maxlen=MAX_LEN)[0]

X_img.append(feature)
X_seq.append(in_seq)
y.append(out_word)

X_img = np.array(X_img)
X_seq = np.array(X_seq)
y = np.array(y)

print("Training samples:", len(y))

image_input = layers.Input(shape=(2048,))
image_dense = layers.Dense(EMBED_DIM, activation="relu")(image_input)
image_repeat = layers.RepeatVector(MAX_LEN)(image_dense)

caption_input = layers.Input(shape=(MAX_LEN,))
caption_embed = layers.Embedding(vocab_size, EMBED_DIM, mask_zero=True)(caption_input)

merged = layers.Add()([image_repeat, caption_embed])
lstm = layers.LSTM(LSTM_UNITS)(merged)
dense = layers.Dense(256, activation="relu")(lstm)
output = layers.Dense(vocab_size, activation="softmax")(dense)

model = Model(inputs=[image_input, caption_input], outputs=output)

model.compile(
optimizer="adam",
loss="sparse_categorical_crossentropy",
metrics=["accuracy"]
)

model.summary()

history = model.fit(
[X_img, X_seq],
y,
batch_size=BATCH_SIZE,
epochs=EPOCHS,
validation_split=0.1
)

def word_for_id(word_id):
return tokenizer.index_word.get(word_id, None)

def generate_caption(image_name):
feature = features[image_name]
caption = "startseq"

for _ in range(MAX_LEN):
seq = tokenizer.texts_to_sequences([caption])[0]
seq = pad_sequences([seq], maxlen=MAX_LEN)

pred = model.predict([np.array([feature]), seq], verbose=0)
pred_id = np.argmax(pred)

word = word_for_id(pred_id)

if word is None:
break

caption += " " + word

if word == "endseq":
break

caption = caption.replace("startseq", "").replace("endseq", "").strip()
return caption

sample_images = list(features.keys())[:5]

plt.figure(figsize=(15, 10))

for i, img_name in enumerate(sample_images):
img_path = os.path.join(IMAGE_DIR, img_name)
img = Image.open(img_path)

pred_caption = generate_caption(img_name)

plt.subplot(1, 5, i + 1)
plt.imshow(img)
plt.axis("off")
plt.title(pred_caption, fontsize=9)

plt.show()

plt.plot(history.history["loss"], label="Training Loss")
plt.plot(history.history["val_loss"], label="Validation Loss")
plt.xlabel("Epoch")
plt.ylabel("Loss")
plt.legend()
plt.show()

model.save("resnet50_lstm_caption_model.h5")
print("Model saved successfully")








Expeiment-8 -BERT text summarizer


import pandas as pd
import torch
from torch.utils.data import Dataset, DataLoader
from transformers import BertTokenizer, EncoderDecoderModel

# ==============================
# CONFIG
# ==============================

DATASET_PATH = "text_summary_dataset.csv"

MAX_INPUT_LEN = 256
MAX_SUMMARY_LEN = 64
BATCH_SIZE = 2
EPOCHS = 3
LR = 5e-5

DEVICE = torch.device("cuda" if torch.cuda.is_available() else "cpu")
print("Using device:", DEVICE)

# ==============================
# LOAD DATASET
# ==============================

data = pd.read_csv(DATASET_PATH)
data = data.dropna()

texts = data["text"].astype(str).tolist()[:500]
summaries = data["summary"].astype(str).tolist()[:500]

print("Total samples:", len(texts))

# ==============================
# LOAD TOKENIZER AND MODEL
# ==============================

tokenizer = BertTokenizer.from_pretrained("bert-base-uncased")

model = EncoderDecoderModel.from_encoder_decoder_pretrained(
"bert-base-uncased",
"bert-base-uncased"
)

model.config.decoder_start_token_id = tokenizer.cls_token_id
model.config.eos_token_id = tokenizer.sep_token_id
model.config.pad_token_id = tokenizer.pad_token_id
model.config.vocab_size = model.config.decoder.vocab_size

model.config.max_length = MAX_SUMMARY_LEN
model.config.min_length = 5
model.config.no_repeat_ngram_size = 2
model.config.early_stopping = True
model.config.length_penalty = 2.0
model.config.num_beams = 4

model.to(DEVICE)

# ==============================
# DATASET CLASS
# ==============================

class SummaryDataset(Dataset):
def __init__(self, texts, summaries, tokenizer):
self.texts = texts
self.summaries = summaries
self.tokenizer = tokenizer

def __len__(self):
return len(self.texts)

def __getitem__(self, idx):
input_enc = self.tokenizer(
self.texts[idx],
max_length=MAX_INPUT_LEN,
padding="max_length",
truncation=True,
return_tensors="pt"
)

target_enc = self.tokenizer(
self.summaries[idx],
max_length=MAX_SUMMARY_LEN,
padding="max_length",
truncation=True,
return_tensors="pt"
)

labels = target_enc["input_ids"].squeeze()
labels[labels == tokenizer.pad_token_id] = -100

return {
"input_ids": input_enc["input_ids"].squeeze(),
"attention_mask": input_enc["attention_mask"].squeeze(),
"labels": labels
}

dataset = SummaryDataset(texts, summaries, tokenizer)

dataloader = DataLoader(
dataset,
batch_size=BATCH_SIZE,
shuffle=True
)

# ==============================
# TRAINING
# ==============================

optimizer = torch.optim.AdamW(model.parameters(), lr=LR)

model.train()

for epoch in range(EPOCHS):
total_loss = 0

for batch in dataloader:
input_ids = batch["input_ids"].to(DEVICE)
attention_mask = batch["attention_mask"].to(DEVICE)
labels = batch["labels"].to(DEVICE)

outputs = model(
input_ids=input_ids,
attention_mask=attention_mask,
labels=labels
)

loss = outputs.loss

optimizer.zero_grad()
loss.backward()
optimizer.step()

total_loss += loss.item()

avg_loss = total_loss / len(dataloader)
print(f"Epoch {epoch + 1}/{EPOCHS} - Loss: {avg_loss:.4f}")

# ==============================
# SUMMARIZATION FUNCTION
# ==============================

def summarize_text(text):
model.eval()

inputs = tokenizer(
text,
max_length=MAX_INPUT_LEN,
padding="max_length",
truncation=True,
return_tensors="pt"
)

input_ids = inputs["input_ids"].to(DEVICE)
attention_mask = inputs["attention_mask"].to(DEVICE)

with torch.no_grad():
summary_ids = model.generate(
input_ids=input_ids,
attention_mask=attention_mask,
max_length=MAX_SUMMARY_LEN,
min_length=5,
num_beams=4,
no_repeat_ngram_size=2,
length_penalty=2.0,
early_stopping=True
)

summary = tokenizer.decode(summary_ids[0], skip_special_tokens=True)
return summary

# ==============================
# TEST
# ==============================

test_text = """
Artificial intelligence is transforming healthcare by improving diagnosis,
treatment planning, patient monitoring, and medical research. AI systems can
analyze large amounts of medical data quickly and help doctors make better
decisions.
"""

generated_summary = summarize_text(test_text)

print("\nInput Text:")
print(test_text)

print("\nGenerated Summary:")
print(generated_summary)

# ==============================
# SAVE MODEL
# ==============================

model.save_pretrained("bert_summarization_model")
tokenizer.save_pretrained("bert_summarization_model")

print("\nModel saved successfully.")





Experiment-9 -Multimodal


import os
import numpy as np
import tensorflow as tf
from tensorflow.keras import layers, Model
from tensorflow.keras.applications import ResNet50
from tensorflow.keras.applications.resnet50 import preprocess_input
from tensorflow.keras.preprocessing.text import Tokenizer
from tensorflow.keras.preprocessing.sequence import pad_sequences

# Load dataset
(x_train, y_train), (x_test, y_test) = tf.keras.datasets.fashion_mnist.load_data()

class_names = [
"t-shirt", "trouser", "pullover", "dress", "coat",
"sandal", "shirt", "sneaker", "bag", "ankle boot"
]

# Take small data for fast training
TRAIN_SAMPLES = 5000
TEST_SAMPLES = 1000

x_train = x_train[:TRAIN_SAMPLES]
y_train = y_train[:TRAIN_SAMPLES]
x_test = x_test[:TEST_SAMPLES]
y_test = y_test[:TEST_SAMPLES]

# Convert grayscale images to RGB
x_train = np.repeat(x_train[..., np.newaxis], 3, axis=-1)
x_test = np.repeat(x_test[..., np.newaxis], 3, axis=-1)

# Create text descriptions
def create_texts(labels):
texts = []
for label in labels:
texts.append("This image contains a " + class_names[int(label)])
return texts

train_texts = create_texts(y_train)
test_texts = create_texts(y_test)

# Tokenize text
MAX_WORDS = 10000
MAX_LEN = 10

tokenizer = Tokenizer(num_words=MAX_WORDS, oov_token="")
tokenizer.fit_on_texts(train_texts)

x_train_text = tokenizer.texts_to_sequences(train_texts)
x_test_text = tokenizer.texts_to_sequences(test_texts)

x_train_text = pad_sequences(x_train_text, maxlen=MAX_LEN, padding="post")
x_test_text = pad_sequences(x_test_text, maxlen=MAX_LEN, padding="post")

# Image input
image_input = layers.Input(shape=(28, 28, 3), name="image_input")

x_img = layers.Resizing(224, 224)(image_input)
x_img = layers.Lambda(lambda x: preprocess_input(tf.cast(x, tf.float32)))(x_img)

resnet = ResNet50(
weights="imagenet",
include_top=False,
input_shape=(224, 224, 3)
)

resnet.trainable = False

x_img = resnet(x_img, training=False)
x_img = layers.GlobalAveragePooling2D()(x_img)
x_img = layers.Dense(256, activation="relu")(x_img)
x_img = layers.Dropout(0.3)(x_img)

# Text input
text_input = layers.Input(shape=(MAX_LEN,), name="text_input")

x_txt = layers.Embedding(input_dim=MAX_WORDS, output_dim=128)(text_input)
x_txt = layers.LSTM(128)(x_txt)
x_txt = layers.Dense(128, activation="relu")(x_txt)
x_txt = layers.Dropout(0.3)(x_txt)

# Combine image and text features
combined = layers.Concatenate()([x_img, x_txt])
combined = layers.Dense(256, activation="relu")(combined)
combined = layers.Dropout(0.4)(combined)

output = layers.Dense(10, activation="softmax")(combined)

model = Model(inputs=[image_input, text_input], outputs=output)

# Compile model
model.compile(
optimizer=tf.keras.optimizers.Adam(learning_rate=0.0001),
loss="sparse_categorical_crossentropy",
metrics=["accuracy"]
)

model.summary()

# Normal input format
train_inputs = {
"image_input": x_train,
"text_input": x_train_text
}

test_inputs = {
"image_input": x_test,
"text_input": x_test_text
}

# Train model
history = model.fit(
train_inputs,
y_train,
validation_data=(test_inputs, y_test),
batch_size=16,
epochs=2
)

# Evaluate model
test_loss, test_acc = model.evaluate(test_inputs, y_test)

print("Test Loss:", test_loss)
print("Test Accuracy:", test_acc)

# Predictions
pred_probs = model.predict(test_inputs)
pred_classes = np.argmax(pred_probs, axis=1)

print("\nSample Predictions:")

for i in range(10):
print("True:", class_names[y_test[i]], "| Predicted:", class_names[pred_classes[i]])
